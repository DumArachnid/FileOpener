// --- OFFLINE SHARDS SYSTEM ---
function processOfflineShards() {
    const now = Date.now();
    const lastSavedTime = parseInt(localStorage.getItem('disguise_last_active_time') || now.toString(), 10);
    const timeAway = now - lastSavedTime;

    if (timeAway > 2000) {
        const cappedTimeAway = Math.min(timeAway, CONFIG.MAX_AWAY_TIME_MS);
        const secondsAway = Math.floor(cappedTimeAway / 1000);
        const shardsEarned = secondsAway * CONFIG.SHARDS_PER_SECOND;

        if (shardsEarned > 0) {
            shards += shardsEarned;
            localStorage.setItem('disguise_shards', shards);
            
            const wasCapped = timeAway > CONFIG.MAX_AWAY_TIME_MS ? " (Max 3hrs)" : "";
            flash(claimRewardBtn, `+${shardsEarned} Offline Shards!${wasCapped}`);
            renderShop();
            shardsDisplay.textContent = shards;
        }
    }
    updateLastActiveTime();
}

function updateLastActiveTime() {
    localStorage.setItem('disguise_last_active_time', Date.now());
}

// --- FLUCTUATING SHOP PRICES ---
function updateShopPrices() {
    const currentMinute = Math.floor(Date.now() / 60000);
    
    shopItems.forEach((item, index) => {
        const itemOffset = (index + 1) * 1.5; 
        const fluctuation = Math.sin((currentMinute * 0.1) + itemOffset);
        const maxVariance = 0.5; 
        
        const priceMultiplier = 1 + (fluctuation * maxVariance);
        let currentPrice = Math.floor(item.baseCost * priceMultiplier);
        
        if (currentPrice < 5) currentPrice = 5;
        
        item.cost = currentPrice;
    });
    
    renderShop();
}

function handleFileOpen(file){
revokeCurrent();
currentFile = file; currentIsText = false; currentIsHTML = false; savedEncoded = null;
editorPanel.classList.add('hidden');
updateActionsVisibility(false);
setStatus(`${file.name} • ${bytesToSize(file.size)}`, 'Not saved');
iframeWarning.classList.remove('show');

if(file.type.startsWith('image/')){
currentBlobURL = URL.createObjectURL(file);
previewArea.innerHTML = `<img src="${currentBlobURL}" alt="${escapeHTML(file.name)}" />`;
updateActionsVisibility(true);
setFontControlsVisible(false);
}
else if(file.type.startsWith('text/') || /\.(html?|js|css|json|md|txt|py|java|c|cpp|csv)$/i.test(file.name)){
file.text().then(text=>{
currentIsText = true; currentIsHTML = /\.html?$/i.test(file.name);
savedEncoded = b64EncodeUnicode(text);
previewArea.innerHTML = `<pre>${escapeHTML(text)}</pre>`;
updateActionsVisibility(true);
setFontControlsVisible(true);
});
} else {
currentBlobURL = URL.createObjectURL(file);
previewArea.innerHTML = `<iframe src="${currentBlobURL}" style="width:100%;height:100%;border:none"></iframe>`;
updateActionsVisibility(true);
setFontControlsVisible(false);
}
}
